Custom Entity Factory

For full control over which entities are included and how their metadata is computed, provide a custom CommitLogEntityFactory:

public class MyEntityFactory implements CommitLogEntityFactory {

    @Override
    public CommitLogEntity getEntity(ObjectId id) {

        // the returned CommitLogEntity defines whether the entity and its individual properties are included
        boolean entityIncluded = !"Painting".equals(id.getEntityName());

        return new CommitLogEntity() {

            @Override
            public boolean isIncluded() {
                return entityIncluded;
            }

            @Override
            public boolean isIncluded(String property) {
                // all the properties of the excluded entity must be excluded as well
                return entityIncluded;
            }

            @Override
            public boolean isConfidential(String property) {
                return false;
            }
        };
    }
}
CoreModule.extend(binder)
        .commitLogEntityFactory(MyEntityFactory.class)
        .addCommitLogListener(AuditListener.class);
